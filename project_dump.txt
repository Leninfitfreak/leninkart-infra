import os
import argparse

def dump_project_to_file(root_dir='.', output_file='project_dump.txt', ignore_dirs=None, ignore_files=None):
    if ignore_dirs is None:
        ignore_dirs = {'.git', '__pycache__', '.idea', '.vscode', 'node_modules', 'build', 'dist', '.gradle'}
    if ignore_files is None:
        ignore_files = {'.DS_Store', 'Thumbs.db'}

    with open(output_file, 'w', encoding='utf-8') as outfile:
        outfile.write(f"# Project Dump - Root: {os.path.abspath(root_dir)}\n\n")

        for current_dir, dirs, files in os.walk(root_dir):
            # Skip ignored directories (in-place modification)
            dirs[:] = [d for d in dirs if d not in ignore_dirs]

            relative_dir = os.path.relpath(current_dir, root_dir)
            if relative_dir != '.':
                outfile.write(f"### DIRECTORY: {relative_dir}/\n\n")

            for file_name in sorted(files):
                if file_name in ignore_files:
                    continue

                file_path = os.path.join(current_dir, file_name)
                relative_path = os.path.relpath(file_path, root_dir)

                outfile.write(f"### FILE: {relative_path}\n")
                outfile.write("```\n")

                try:
                    with open(file_path, 'r', encoding='utf-8') as f:
                        content = f.read()
                    outfile.write(content.rstrip() + "\n")  # Ensure trailing newline
                except UnicodeDecodeError:
                    outfile.write("<BINARY OR NON-UTF8 FILE - CONTENT SKIPPED>\n")
                except PermissionError:
                    outfile.write("<PERMISSION DENIED - CONTENT SKIPPED>\n")
                except Exception as e:
                    outfile.write(f"<ERROR READING FILE: {str(e)}>\n")

                outfile.write("```\n\n")

    print(f"Project dump completed! File saved as '{output_file}' in the current directory.")
    print("Share the content of this file with me — I'll analyze every line and fix LeninKart!")

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Dump entire project file structure and content to a single text file")
    parser.add_argument('-o', '--output', default='project_dump.txt', help='Output file name (default: project_dump.txt)')
    parser.add_argument('-r', '--root', default='.', help='Root directory to scan (default: current directory)')

    args = parser.parse_args()

    dump_project_to_file(root_dir=args.root, output_file=args.output)

#!/usr/bin/env python3
"""
LeninKart Kubernetes Diagnostic Script
=======================================
Analyzes the entire Kubernetes deployment to identify issues.
Run this and share the output!
"""

import subprocess
import json
import sys
from datetime import datetime

# ANSI color codes
GREEN = '\033[92m'
RED = '\033[91m'
YELLOW = '\033[93m'
BLUE = '\033[94m'
RESET = '\033[0m'

def run_cmd(cmd):
    """Run a kubectl command and return output"""
    try:
        result = subprocess.run(
            cmd,
            shell=True,
            capture_output=True,
            text=True,
            timeout=30
        )
        return result.stdout, result.stderr, result.returncode
    except Exception as e:
        return "", str(e), 1

def log(msg, color=RESET):
    """Print colored log message"""
    print(f"{color}[DIAG] {msg}{RESET}")

def section(title):
    """Print section header"""
    print(f"\n{BLUE}{'='*70}{RESET}")
    print(f"{BLUE}{title}{RESET}")
    print(f"{BLUE}{'='*70}{RESET}\n")

def check_pods():
    """Check pod status"""
    section("1. POD STATUS")
    stdout, stderr, code = run_cmd("kubectl get pods -n dev -o wide")
    print(stdout)
    
    # Parse and analyze
    lines = stdout.strip().split('\n')[1:]  # Skip header
    issues = []
    for line in lines:
        parts = line.split()
        if len(parts) < 3:
            continue
        name, ready, status = parts[0], parts[1], parts[2]
        
        if status != "Running":
            issues.append(f"{RED}✗ {name}: {status}{RESET}")
        elif ready != "1/1":
            issues.append(f"{YELLOW}⚠ {name}: Ready {ready}{RESET}")
        else:
            print(f"{GREEN}✓ {name}: Running and Ready{RESET}")
    
    if issues:
        print("\nISSUES FOUND:")
        for issue in issues:
            print(issue)
    
    return len(issues) == 0

def check_services():
    """Check service configuration"""
    section("2. SERVICE STATUS")
    stdout, _, _ = run_cmd("kubectl get svc -n dev -o wide")
    print(stdout)
    
    section("2a. SERVICE ENDPOINTS")
    stdout, _, _ = run_cmd("kubectl get endpoints -n dev")
    print(stdout)
    
    # Check for services with no endpoints
    lines = stdout.strip().split('\n')[1:]
    issues = []
    for line in lines:
        parts = line.split()
        if len(parts) < 2:
            continue
        name, endpoints = parts[0], parts[1]
        
        if endpoints == "<none>" or not endpoints:
            issues.append(f"{RED}✗ {name}: NO ENDPOINTS{RESET}")
        else:
            print(f"{GREEN}✓ {name}: {endpoints}{RESET}")
    
    if issues:
        print("\nSERVICE ISSUES:")
        for issue in issues:
            print(issue)
    
    return len(issues) == 0

def check_service_selectors():
    """Check if service selectors match pod labels"""
    section("3. SERVICE SELECTOR VALIDATION")
    
    services = ["leninkart-frontend", "leninkart-product-service", "leninkart-order-service"]
    all_match = True
    
    for svc in services:
        log(f"Checking {svc}...")
        
        # Get service selector
        stdout, _, _ = run_cmd(f"kubectl get svc {svc} -n dev -o jsonpath='{{.spec.selector}}'")
        print(f"  Service selector: {stdout}")
        
        # Get matching pods
        # Extract selector pairs from JSON
        try:
            selector_dict = json.loads(stdout.replace("'", '"'))
            selector_parts = [f"{k}={v}" for k, v in selector_dict.items()]
            selector_str = ",".join(selector_parts)
            
            stdout, _, _ = run_cmd(f"kubectl get pods -n dev -l {selector_str} -o name")
            matching_pods = stdout.strip().split('\n') if stdout.strip() else []
            
            if matching_pods and matching_pods[0]:
                print(f"{GREEN}  ✓ Matching pods: {len(matching_pods)}{RESET}")
                for pod in matching_pods[:3]:  # Show first 3
                    print(f"    - {pod}")
            else:
                print(f"{RED}  ✗ NO MATCHING PODS FOUND!{RESET}")
                all_match = False
        except Exception as e:
            print(f"{RED}  ✗ Error parsing selector: {e}{RESET}")
            all_match = False
        
        print()
    
    return all_match

def check_ingress():
    """Check ingress configuration"""
    section("4. INGRESS STATUS")
    stdout, _, _ = run_cmd("kubectl get ingress -n dev")
    print(stdout)
    
    section("4a. INGRESS DETAILS")
    stdout, _, _ = run_cmd("kubectl describe ingress leninkart-ingress -n dev")
    print(stdout)

def check_ingress_controller():
    """Check ingress controller status"""
    section("5. INGRESS CONTROLLER")
    stdout, _, _ = run_cmd("kubectl get pods -n ingress-nginx")
    print(stdout)
    
    section("5a. INGRESS CONTROLLER LOGS (Last 30 lines)")
    stdout, _, _ = run_cmd("kubectl logs -n ingress-nginx -l app.kubernetes.io/name=ingress-nginx --tail=30")
    
    # Highlight warnings
    for line in stdout.split('\n'):
        if "does not have any active Endpoint" in line:
            print(f"{RED}{line}{RESET}")
        elif "WARN" in line or "ERROR" in line:
            print(f"{YELLOW}{line}{RESET}")
        else:
            print(line)

def check_connectivity():
    """Test internal connectivity"""
    section("6. CONNECTIVITY TESTS")
    
    services = [
        ("Product Service", "leninkart-product-service.dev.svc.cluster.local:8081/api/products"),
        ("Order Service", "leninkart-order-service.dev.svc.cluster.local:8080/api/orders"),
        ("Frontend", "leninkart-frontend.dev.svc.cluster.local:80"),
    ]
    
    for name, url in services:
        log(f"Testing {name}...")
        cmd = f"kubectl run test-curl-{name.lower().replace(' ', '-')} --rm -i --restart=Never --image=curlimages/curl --timeout=10s -- curl -s -o /dev/null -w '%{{http_code}}' http://{url}"
        stdout, stderr, code = run_cmd(cmd)
        
        if "200" in stdout:
            print(f"{GREEN}  ✓ {name}: HTTP 200 OK{RESET}")
        elif stdout.strip():
            print(f"{YELLOW}  ⚠ {name}: HTTP {stdout.strip()}{RESET}")
        else:
            print(f"{RED}  ✗ {name}: FAILED - {stderr[:100]}{RESET}")
        print()

def check_backend_health():
    """Check backend service health"""
    section("7. BACKEND SERVICE HEALTH")
    
    # Check product-service logs
    log("Product Service Recent Logs:")
    stdout, _, _ = run_cmd("kubectl logs -n dev -l app=product-service --tail=10")
    print(stdout[-500:] if len(stdout) > 500 else stdout)
    
    print()
    
    # Check order-service logs
    log("Order Service Recent Logs:")
    stdout, _, _ = run_cmd("kubectl logs -n dev -l app.kubernetes.io/name=order-service --tail=10")
    print(stdout[-500:] if len(stdout) > 500 else stdout)

def check_frontend_image():
    """Check frontend image and age"""
    section("8. FRONTEND IMAGE INFO")
    stdout, _, _ = run_cmd("kubectl get pods -n dev -l app=frontend -o jsonpath='{.items[0].spec.containers[0].image}'")
    print(f"Image: {stdout}")
    
    stdout, _, _ = run_cmd("kubectl get pods -n dev -l app=frontend -o jsonpath='{.items[0].metadata.creationTimestamp}'")
    print(f"Pod Created: {stdout}")
    
    stdout, _, _ = run_cmd("kubectl get pods -n dev -l app=frontend -o jsonpath='{.items[0].status.containerStatuses[0].restartCount}'")
    print(f"Restart Count: {stdout}")

def main():
    """Main diagnostic routine"""
    print(f"\n{GREEN}LeninKart Kubernetes Diagnostic Script{RESET}")
    print(f"{GREEN}Started: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}{RESET}\n")
    
    results = {}
    
    results['pods'] = check_pods()
    results['services'] = check_services()
    results['selectors'] = check_service_selectors()
    
    check_ingress()
    check_ingress_controller()
    
    # Skip connectivity tests if basic checks failed
    if results['pods'] and results['services']:
        check_connectivity()
    else:
        section("6. CONNECTIVITY TESTS")
        print(f"{YELLOW}Skipped - Fix pods and services first{RESET}")
    
    check_backend_health()
    check_frontend_image()
    
    # Summary
    section("SUMMARY")
    all_good = all(results.values())
    
    if all_good:
        print(f"{GREEN}✓ All basic checks passed!{RESET}")
        print(f"{YELLOW}If app still not working, check ingress controller and connectivity tests above.{RESET}")
    else:
        print(f"{RED}✗ Issues found:{RESET}")
        for check, passed in results.items():
            status = f"{GREEN}✓{RESET}" if passed else f"{RED}✗{RESET}"
            print(f"  {status} {check.upper()}")
    
    print(f"\n{BLUE}Diagnostic complete!{RESET}\n")

if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        print(f"\n{YELLOW}Diagnostic interrupted by user{RESET}")
        sys.exit(1)
    except Exception as e:
        print(f"\n{RED}Error: {e}{RESET}")
        sys.exit(1)
```
