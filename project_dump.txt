import os
import argparse

def dump_project_to_file(root_dir='.', output_file='project_dump.txt', ignore_dirs=None, ignore_files=None):
    if ignore_dirs is None:
        ignore_dirs = {'.git', '__pycache__', '.idea', '.vscode', 'node_modules', 'build', 'dist', '.gradle'}
    if ignore_files is None:
        ignore_files = {'.DS_Store', 'Thumbs.db'}

    with open(output_file, 'w', encoding='utf-8') as outfile:
        outfile.write(f"# Project Dump - Root: {os.path.abspath(root_dir)}\n\n")

        for current_dir, dirs, files in os.walk(root_dir):
            # Skip ignored directories (in-place modification)
            dirs[:] = [d for d in dirs if d not in ignore_dirs]

            relative_dir = os.path.relpath(current_dir, root_dir)
            if relative_dir != '.':
                outfile.write(f"### DIRECTORY: {relative_dir}/\n\n")

            for file_name in sorted(files):
                if file_name in ignore_files:
                    continue

                file_path = os.path.join(current_dir, file_name)
                relative_path = os.path.relpath(file_path, root_dir)

                outfile.write(f"### FILE: {relative_path}\n")
                outfile.write("```\n")

                try:
                    with open(file_path, 'r', encoding='utf-8') as f:
                        content = f.read()
                    outfile.write(content.rstrip() + "\n")  # Ensure trailing newline
                except UnicodeDecodeError:
                    outfile.write("<BINARY OR NON-UTF8 FILE - CONTENT SKIPPED>\n")
                except PermissionError:
                    outfile.write("<PERMISSION DENIED - CONTENT SKIPPED>\n")
                except Exception as e:
                    outfile.write(f"<ERROR READING FILE: {str(e)}>\n")

                outfile.write("```\n\n")

    print(f"Project dump completed! File saved as '{output_file}' in the current directory.")
    print("Share the content of this file with me — I'll analyze every line and fix LeninKart!")

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Dump entire project file structure and content to a single text file")
    parser.add_argument('-o', '--output', default='project_dump.txt', help='Output file name (default: project_dump.txt)')
    parser.add_argument('-r', '--root', default='.', help='Root directory to scan (default: current directory)')

    args = parser.parse_args()

    dump_project_to_file(root_dir=args.root, output_file=args.output)

#!/usr/bin/env python3
"""
LeninKart Kubernetes Debug Script
Checks the actual state of your cluster and identifies issues
"""

import subprocess
import json
import sys

class K8sDebugger:
    def __init__(self, namespace="dev"):
        self.namespace = namespace
        self.issues = []
        
    def run_cmd(self, cmd):
        """Run kubectl command and return output"""
        try:
            result = subprocess.run(
                cmd, 
                shell=True, 
                capture_output=True, 
                text=True, 
                timeout=10
            )
            return result.stdout.strip(), result.returncode
        except Exception as e:
            return f"Error: {str(e)}", 1
    
    def check_pods(self):
        """Check if all pods are running"""
        print("\n" + "="*60)
        print("🔍 CHECKING PODS")
        print("="*60)
        
        output, code = self.run_cmd(f"kubectl get pods -n {self.namespace}")
        print(output)
        
        if code != 0:
            self.issues.append("❌ Failed to get pods - is kubectl configured?")
            return False
        
        # Check for non-running pods
        lines = output.split('\n')[1:]  # Skip header
        for line in lines:
            if line and 'Running' not in line:
                self.issues.append(f"❌ Pod not running: {line.split()[0]}")
        
        return True
    
    def check_services(self):
        """Check if services exist and have endpoints"""
        print("\n" + "="*60)
        print("🔍 CHECKING SERVICES")
        print("="*60)
        
        services = [
            "leninkart-product-service",
            "leninkart-order-service", 
            "leninkart-frontend"
        ]
        
        for svc in services:
            output, code = self.run_cmd(f"kubectl get svc {svc} -n {self.namespace} -o wide")
            
            if code != 0:
                print(f"❌ {svc} NOT FOUND")
                self.issues.append(f"❌ Service {svc} does not exist")
            else:
                print(f"✓ {svc}")
                print(f"  {output.split(chr(10))[1]}")
                
                # Check endpoints
                ep_out, ep_code = self.run_cmd(f"kubectl get endpoints {svc} -n {self.namespace}")
                if 'none' in ep_out.lower():
                    self.issues.append(f"⚠️  {svc} has no endpoints (no pods selected)")
        
        return True
    
    def check_ingress(self):
        """Check ingress configuration"""
        print("\n" + "="*60)
        print("🔍 CHECKING INGRESS")
        print("="*60)
        
        output, code = self.run_cmd(f"kubectl get ingress -n {self.namespace}")
        print(output)
        
        if code != 0 or 'leninkart-ingress' not in output:
            self.issues.append("❌ Ingress not found")
            return False
        
        # Get detailed ingress info
        detail, _ = self.run_cmd(f"kubectl describe ingress leninkart-ingress -n {self.namespace}")
        print("\nIngress Details:")
        
        # Extract paths
        for line in detail.split('\n'):
            if 'Path:' in line or 'Backend:' in line:
                print(f"  {line.strip()}")
        
        return True
    
    def test_service_directly(self):
        """Test if we can reach services directly"""
        print("\n" + "="*60)
        print("🔍 TESTING DIRECT SERVICE ACCESS")
        print("="*60)
        
        print("\nTesting product-service...")
        print("Run this command in another terminal:")
        print(f"  kubectl port-forward -n {self.namespace} svc/leninkart-product-service 8081:8081")
        print("  Then visit: http://localhost:8081/api/products")
        
        return True
    
    def check_pod_logs(self):
        """Get recent logs from backend services"""
        print("\n" + "="*60)
        print("🔍 CHECKING POD LOGS (last 10 lines)")
        print("="*60)
        
        services = [
            ("product-service", "app=product-service"),
            ("order-service", "app.kubernetes.io/name=order-service")
        ]
        
        for name, label in services:
            print(f"\n--- {name} ---")
            output, code = self.run_cmd(
                f"kubectl logs -n {self.namespace} -l {label} --tail=10 2>&1"
            )
            if code == 0 and output:
                print(output)
            else:
                print(f"No logs or pod not found")
    
    def suggest_fixes(self):
        """Provide actionable fixes based on issues found"""
        print("\n" + "="*60)
        print("💡 DIAGNOSIS & FIXES")
        print("="*60)
        
        if not self.issues:
            print("\n✅ No obvious issues found!")
            print("\nBut you're still getting 404s. This means:")
            print("1. Services might not be registered correctly")
            print("2. Ingress routing is misconfigured")
            print("3. Backend services aren't responding on expected paths")
            print("\nQuick test:")
            print(f"  kubectl port-forward -n {self.namespace} svc/leninkart-product-service 8081:8081")
            print("  curl http://localhost:8081/api/products")
            print("\nIf that works, the issue is in the ingress routing.")
        else:
            print("\n🔴 Issues Found:")
            for issue in self.issues:
                print(f"  {issue}")
        
        print("\n" + "="*60)
        print("RECOMMENDED ACTIONS:")
        print("="*60)
        print("1. Run: kubectl port-forward -n dev svc/leninkart-product-service 8081:8081")
        print("   Then test: curl http://localhost:8081/api/products")
        print()
        print("2. If #1 fails: Check pod logs for errors")
        print("   kubectl logs -n dev -l app=product-service --tail=50")
        print()
        print("3. If #1 works: Issue is in ingress routing")
        print("   Run the fix_infra.py script to fix ingress")
        print()
        print("4. Ensure minikube tunnel is running:")
        print("   minikube tunnel")
        print("="*60)
    
    def run_full_check(self):
        """Run all checks"""
        print("\n🚀 LeninKart Kubernetes Debugger")
        print(f"   Namespace: {self.namespace}")
        
        self.check_pods()
        self.check_services()
        self.check_ingress()
        self.test_service_directly()
        self.check_pod_logs()
        self.suggest_fixes()

if __name__ == "__main__":
    namespace = sys.argv[1] if len(sys.argv) > 1 else "dev"
    
    debugger = K8sDebugger(namespace)
    debugger.run_full_check()
```
